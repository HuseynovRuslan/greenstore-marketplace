function FooterBottomBar() {
  const activeSocialProfiles = socialProfiles.filter(
    (profile): profile is SocialLink & { href: string } =>
      typeof profile.href === "string" && /^https:\/\//i.test(profile.href),
  );

  return (
    <div className="mt-10 flex flex-col items-start gap-5 border-t border-mint/20 pt-6 text-[13px] font-normal text-mint xl:flex-row xl:items-center xl:justify-between xl:gap-7">
      <p className="shrink-0 leading-5">© {new Date().getFullYear()} GreenStore. Bütün hüquqlar qorunur.</p>

      <nav aria-label="Hüquqi məlumatlar">
        <ul className="flex flex-wrap items-center gap-x-6 gap-y-3">
          {legalLinks.map((link) => (
            <li key={link.href}>
              <Link
                href={link.href}
                className={`inline-block rounded-sm leading-5 text-mint hover:text-white hover:underline hover:underline-offset-4 ${darkFocus}`}
              >
                {link.label}
              </Link>
            </li>
          ))}
        </ul>
      </nav>

      {activeSocialProfiles.length > 0 && (
        <nav aria-label="GreenStore sosial şəbəkələri">
          <ul className="flex items-center gap-2.5">
            {activeSocialProfiles.map((social) => (
              <li key={social.provider}>
                <a
                  href={social.href}
                  target="_blank"
                  rel="noopener noreferrer"
                  aria-label={`${social.ariaLabel} — yeni pəncərədə açılır`}
                  className={`grid size-10 place-items-center rounded-xl border border-mint/30 text-mint transition-colors hover:border-mint hover:bg-mint/10 hover:text-white ${darkFocus}`}
                >
                  <SocialBrandIcon provider={social.provider} />
                </a>
              </li>
            ))}
          </ul>
        </nav>
      )}
    </div>
  );
}
